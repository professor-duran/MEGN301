\chapter*{Glossary of Acronyms and Key Terms}
\addcontentsline{toc}{chapter}{Glossary}
\markboth{Glossary of Acronyms and Key Terms}{}

\begin{description}[leftmargin=3.5cm, labelwidth=3cm]
\item[\textbf{ABET}]\index{ABET outcomes} Accreditation Board for Engineering and Technology
\item[\textbf{ADC}] Analog-to-Digital Converter
\item[\textbf{BLE}] Bluetooth Low Energy
\item[\textbf{BLDC}] Brushless Direct Current (motor type)
\item[\textbf{BOM}] Bill of Materials; complete list of every component
\item[\textbf{BSPP}] British Standard Pipe Parallel (thread standard)
\item[\textbf{BSPT}] British Standard Pipe Taper (thread standard)
\item[\textbf{CAN}] Controller Area Network (communication bus)
\item[\textbf{CDR}] Critical Design Review: ``Will this design work?''
\item[\textbf{ConOps}] Concept of Operations; stakeholder-facing system description
\item[\textbf{COTS}] Commercial Off-The-Shelf (product)
\item[\textbf{DFA}] Design for Assembly
\item[\textbf{DFM}]\index{Design for Manufacturability (DFM)} Design for Manufacturability
\item[\textbf{DR}] Discrepancy Report
\item[\textbf{EMC/EMI}] Electromagnetic Compatibility / Interference
\item[\textbf{ESC}] Electronic Speed Controller (for BLDC motors)
\item[\textbf{FDR}] Final Design Review: ``How well did the design work?''
\item[\textbf{FEA}] Finite Element Analysis
\item[\textbf{FMEA}] Failure Modes and Effects Analysis
\item[\textbf{FMECA}] Failure Modes, Effects, and Criticality Analysis
\item[\textbf{FDM}] Fused Deposition Modeling (3D printing)
\item[\textbf{GPIO}] General Purpose Input/Output
\item[\textbf{I2C}] Inter-Integrated Circuit (2-wire serial bus)
\item[\textbf{ICD}] Interface Control Document
\item[\textbf{INCOSE}] International Council on Systems Engineering
\item[\textbf{IP (rating)}] Ingress Protection (e.g., IP67 = dust-tight, 1\,m submersion)
\item[\textbf{LDO}] Low Drop-Out (voltage regulator)
\item[\textbf{MTBF}] Mean Time Between Failures
\item[\textbf{MTTR}] Mean Time To Repair
\item[\textbf{MCU}] Microcontroller Unit
\item[\textbf{MoSCoW}] Must / Should / Could / Won't (prioritization)
\item[\textbf{NEMA}] National Electrical Manufacturers Association (motor frame sizes)
\item[\textbf{NPT}] National Pipe Thread (tapered, U.S.\ standard)
\item[\textbf{NPS}] Nominal Pipe Size
\item[\textbf{PDR}] Preliminary Design Review: ``Should this design work?''
\item[\textbf{PWM}] Pulse Width Modulation
\item[\textbf{RPN}] Risk Priority Number (Severity $\times$ Occurrence $\times$ Detection)
\item[\textbf{RTM}] Requirements Traceability Matrix; links each requirement to its parent need and forward to its verification method. Tracks \emph{what} must be verified.
\item[\textbf{SDRD}] System Design Requirements Document
\item[\textbf{SPI (comms)}] Serial Peripheral Interface (4-wire bus)
\item[\textbf{SPI (mold)}] Society of the Plastics Industry (mold finish standards)
\item[\textbf{TAID}] Test, Analysis, Inspection, Demonstration (verification methods)
\item[\textbf{THT}] Through-Hole Technology (component mounting)
\item[\textbf{SMD/SMT}] Surface-Mount Device / Technology
\item[\textbf{TVS}] Transient Voltage Suppressor (protection diode)
\item[\textbf{UART}] Universal Asynchronous Receiver/Transmitter
\item[\textbf{UNC}] Unified National Coarse (thread standard)
\item[\textbf{UNF}] Unified National Fine (thread standard)
\item[\textbf{V\&V}] Verification and Validation
\item[\textbf{VCRM}] Verification Cross-Reference Matrix; extends the RTM by recording test status (Planned/In Progress/PASS/FAIL/PARTIAL) at each review milestone. Tracks \emph{whether} each requirement has been verified.
\end{description}

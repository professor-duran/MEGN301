\chapter*{Glossary of Acronyms and Key Terms}
\addcontentsline{toc}{chapter}{Glossary}
\markboth{Glossary of Acronyms and Key Terms}{}

\begin{description}[style=nextline, leftmargin=3.5cm, labelwidth=3cm, font=\normalfont\bfseries]
\item[ABET\index{ABET outcomes}] Accreditation Board for Engineering and Technology
\item[ADC] Analog-to-Digital Converter
\item[BLE] Bluetooth Low Energy
\item[BLDC] Brushless Direct Current (motor type)
\item[BOM] Bill of Materials; complete list of every component
\item[BSPP] British Standard Pipe Parallel (thread standard)
\item[BSPT] British Standard Pipe Taper (thread standard)
\item[CAN] Controller Area Network (communication bus)
\item[CDR] Critical Design Review: ``Will this design work?''
\item[ConOps] Concept of Operations; stakeholder-facing system description
\item[COTS] Commercial Off-The-Shelf (product)
\item[DFA] Design for Assembly
\item[DFM\index{Design for Manufacturability (DFM)}] Design for Manufacturability
\item[DR] Discrepancy Report
\item[EMC/EMI] Electromagnetic Compatibility / Interference
\item[ESC] Electronic Speed Controller (for BLDC motors)
\item[FDR] Final Design Review: ``How well did the design work?''
\item[FEA] Finite Element Analysis
\item[FMEA] Failure Modes and Effects Analysis
\item[FMECA] Failure Modes, Effects, and Criticality Analysis
\item[FDM] Fused Deposition Modeling (3D printing)
\item[GPIO] General Purpose Input/Output
\item[I2C] Inter-Integrated Circuit (2-wire serial bus)
\item[ICD] Interface Control Document
\item[INCOSE] International Council on Systems Engineering
\item[IP (rating)] Ingress Protection (e.g., IP67 = dust-tight, 1\,m submersion)
\item[LDO] Low Drop-Out (voltage regulator)
\item[MTBF] Mean Time Between Failures
\item[MTTR] Mean Time To Repair
\item[MCU] Microcontroller Unit
\item[MoSCoW] Must / Should / Could / Won't (prioritization)
\item[NEMA] National Electrical Manufacturers Association (motor frame sizes)
\item[NPT] National Pipe Thread (tapered, U.S.\ standard)
\item[NPS] Nominal Pipe Size
\item[PDR] Preliminary Design Review: ``Should this design work?''
\item[PWM] Pulse Width Modulation
\item[RPN] Risk Priority Number (Severity $\times$ Occurrence $\times$ Detection)
\item[RTM] Requirements Traceability Matrix; links each requirement to its parent need and forward to its verification method. Tracks \emph{what} must be verified.
\item[SDRD] System Design Requirements Document
\item[SPI (comms)] Serial Peripheral Interface (4-wire bus)
\item[SPI (mold)] Society of the Plastics Industry (mold finish standards)
\item[TAID] Test, Analysis, Inspection, Demonstration (verification methods)
\item[THT] Through-Hole Technology (component mounting)
\item[SMD/SMT] Surface-Mount Device / Technology
\item[TVS] Transient Voltage Suppressor (protection diode)
\item[UART] Universal Asynchronous Receiver/Transmitter
\item[UNC] Unified National Coarse (thread standard)
\item[UNF] Unified National Fine (thread standard)
\item[V\&V] Verification and Validation
\item[VCRM] Verification Cross-Reference Matrix; extends the RTM by recording test status (Planned/In Progress/PASS/FAIL/PARTIAL) at each review milestone. Tracks \emph{whether} each requirement has been verified.
\end{description}

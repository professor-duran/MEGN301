% fig_pcb_design_workflow.tex — Six-stage PCB design workflow
\begin{tikzpicture}[>=Stealth,
    wfstage/.style={rectangle, rounded corners=4pt, draw=#1, fill=#1!10,
        text width=2.0cm, minimum height=1.6cm, align=center,
        font=\scriptsize\sffamily\bfseries, line width=0.8pt, inner sep=4pt},
    detail/.style={font=\tiny\sffamily, color=MinesBlue!70!black, text width=2.0cm, align=center},
    stagenum/.style={circle, draw=#1, fill=#1, text=white,
        minimum size=0.45cm, font=\tiny\sffamily\bfseries, inner sep=0pt}
]

% Stage nodes
\node[wfstage=MinesBlue]   (s1) at (0,0)     {Schematic\\Capture};
\node[wfstage=AccentTeal]  (s2) at (2.8,0)   {Component\\Selection};
\node[wfstage=WarnOrange]  (s3) at (5.6,0)   {Board\\Layout};
\node[wfstage=diagPurple]  (s4) at (8.4,0)   {Design\\Review};
\node[wfstage=diagGreen]   (s5) at (11.2,0)  {Fab +\\Assembly};
\node[wfstage=diagYellow!80!black] (s6) at (14.0,0)  {Bring-Up\\+ Test};

% Stage numbers
\node[stagenum=MinesBlue]   at ([xshift=-0.7cm, yshift=0.55cm]s1.north west) {1};
\node[stagenum=AccentTeal]  at ([xshift=-0.7cm, yshift=0.55cm]s2.north west) {2};
\node[stagenum=WarnOrange]  at ([xshift=-0.7cm, yshift=0.55cm]s3.north west) {3};
\node[stagenum=diagPurple]  at ([xshift=-0.7cm, yshift=0.55cm]s4.north west) {4};
\node[stagenum=diagGreen]   at ([xshift=-0.7cm, yshift=0.55cm]s5.north west) {5};
\node[stagenum=diagYellow!80!black] at ([xshift=-0.7cm, yshift=0.55cm]s6.north west) {6};

% Flow arrows
\draw[flow=MinesBlue]  (s1) -- (s2);
\draw[flow=AccentTeal] (s2) -- (s3);
\draw[flow=WarnOrange] (s3) -- (s4);
\draw[flow=diagPurple] (s4) -- (s5);
\draw[flow=diagGreen]  (s5) -- (s6);

% Detail annotations below
\node[detail, below=0.3cm of s1] {Draw circuit,\\net connections,\\component symbols};
\node[detail, below=0.3cm of s2] {Map to footprints,\\verify availability,\\create BOM};
\node[detail, below=0.3cm of s3] {Place parts,\\route traces,\\copper fills};
\node[detail, below=0.3cm of s4] {Peer review,\\DRC, verify\\footprints, Gerbers};
\node[detail, below=0.3cm of s5] {Generate Gerbers,\\order boards,\\solder components};
\node[detail, below=0.3cm of s6] {Inspect, smoke test,\\functional test,\\debug};

% Tool annotation
\node[font=\tiny\sffamily\itshape, color=MinesSilver, text width=6cm, align=center]

\draw[->, line width=0.8pt, color=diagRed, dashed]
    (s4.south) -- ++(0,-1.6) -| node[below, pos=0.25, font=\tiny\sffamily\itshape, color=diagRed]{fix errors} (s3.south);

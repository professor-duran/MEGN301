\begin{figure}[htbp]

\begin{tikzpicture}[

% === Power Supply Column ===

\draw[conn=AccentTeal] (psu1) -- (psu2);
\draw[conn=AccentTeal] (psu2) -- (psu3);
\draw[conn=AccentTeal] (psu3) -- (psu4);

\node[step=MinesBlue, below=0.6cm of mot] (mot1) {Calculate load\\torque \& speed};
\node[step=MinesBlue, below=0.5cm of mot1] (mot2) {Select motor type\\(DC/BLDC/stepper)};

\draw[conn=MinesBlue] (mot) -- (mot1);
\draw[conn=MinesBlue] (mot1) -- (mot2);
\draw[conn=MinesBlue] (mot2) -- (mot3);
\draw[conn=MinesBlue] (mot3) -- (mot4);

\node[step=WarnOrange, below=0.5cm of pump2] (pump3) {Select pump type\\(see matrix)};
\node[step=WarnOrange, below=0.5cm of pump3] (pump4) {Verify operating\\point on curve};

% === Integration Verification ===
